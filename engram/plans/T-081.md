# T-081 — Reparar el XML de Rekordbox y exportar un XML para importar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Con la fuente Rekordbox (XML exportado), poder reparar los enlaces rotos de las pistas buscándolas en una carpeta, de modo que la exportación numerada copie TODAS las canciones de la lista, y poder exportar las playlists marcadas como un `rekordbox.xml` nuevo (con las rutas reparadas) para importarlo en Rekordbox como lista nueva.

**Architecture:** List Buddy nunca escribe archivos de Rekordbox ni el XML del usuario (ADR-004 sigue en pie). Las reparaciones se guardan en un archivo propio `rekordbox_fixes.json` (URI `Location` original → ruta nueva) en la carpeta de datos de la app; `rekordbox_xml.open_library` las aplica al cargar, así vista previa, reproducción y exportación ven la ruta reparada sin más cambios. La búsqueda reutiliza `relocate_core.BaseRelocateWorker` (mismo matching, modal y auto-resolución que Traktor) con un worker nuevo cuyo «aplicar» solo anota en ese archivo. La exportación a XML escribe un archivo nuevo (formato oficial `xml_format_list.pdf`) con solo las playlists marcadas, copiando cada `TRACK` original (cues, beatgrid, metadatos) y reemplazando `Location` por la ruta reparada. De paso, la configuración y las reparaciones pasan a una carpeta de datos del usuario cuando la app está empaquetada (hoy `.rbe_settings.ini` se escribe dentro del bundle, que es de solo lectura y rompe la firma).

**Tech Stack:** Python 3.14, PySide6 6.11.1, `xml.etree.ElementTree`, `urllib.parse`, `json`, pytest (+xdist).

**Spec:** pedido del PO en sesión 2026-10-14: «armá la app para que pueda reparar links a canciones del xml para que al exportarlas ordenadamente viajen todos los archivos … y agregá la opción de exportar un xml nuevo para poder importarlo luego en rekordbox … exportar solo la lista y luego importar una lista entera a rekordbox fixeada». from __future__ import annotations

def data_dir() -> Path:
    """Carpeta de datos de la app: fuera del bundle, escribible y por usuario."""
    if sys.platform == "darwin":
        d = Path.home() / "Library" / "Application Support" / "listBuddy"
    elif sys.platform.startswith("win"):

    else:
        d = Path(os.environ.get("XDG_CONFIG_HOME") or Path.home() / ".config") / "listBuddy"

def default_settings_path(frozen: bool | None = None) -> Path:
    """Empaquetada: carpeta de datos (el bundle es de solo lectura y está firmado).
    Desde el código: `.rbe_settings.ini` junto al código, como siempre."""

```python
"""
rekordbox_fixes.py
------------------
Reparaciones de enlaces de Rekordbox hechas en List Buddy (T-081). Se guardan
en un archivo propio de la app — nunca en el XML ni en Rekordbox (ADR-004):
URI `Location` original del XML → ruta nueva en disco.
"""
from __future__ import annotations

import json
import logging
import os
from pathlib import Path

_log = logging.getLogger("listBuddy")
_VERSION = 1


def fixes_path_for(settings_file: Path) -> Path:
    return settings_file.parent / "rekordbox_fixes.json"


class FixStore:
    def __init__(self, path: Path) -> None:
        self.path = path

    def load(self) -> dict[str, str]:
        try:
            data = json.loads(self.path.read_text(encoding="utf-8"))
        except FileNotFoundError:
            return {}
        except (OSError, ValueError) as e:
            _log.warning("Reparaciones de Rekordbox ilegibles (%s): %s", self.path, e)
            return {}
        fixes = data.get("fixes") if isinstance(data, dict) else None
        if not isinstance(fixes, dict):
            return {}
        return {str(k): str(v) for k, v in fixes.items()}

    def save(self, mapping: dict[str, str]) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        tmp = self.path.with_name(self.path.name + ".tmp")
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump({"version": _VERSION, "fixes": mapping}, f, ensure_ascii=False, indent=1)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, self.path)

    def merge(self, new: dict[str, str]) -> dict[str, str]:
        merged = self.load()
        merged.update(new)
        self.save(merged)
        return merged

    def clear(self) -> None:
        self.path.unlink(missing_ok=True)

    def count(self) -> int:
        return len(self.load())
```

def path_to_location(path: PurePath) -> str:
    """Ruta local → URI `file://localhost/...` (inversa de `location_to_path`)."""
    posix = path.as_posix()
    if not posix.startswith("/"):

```python
"""
rekordbox_xml_relocate.py
-------------------------
T-081: reparar enlaces rotos de la fuente Rekordbox (XML exportado). Mismo
loop que Traktor (relocate_core.BaseRelocateWorker: indexado, matching,
modal, auto-resolución), pero «aplicar» solo anota la ruta nueva en el
almacén propio de List Buddy (rekordbox_fixes.json). Nunca escribe el XML
ni ningún archivo de Rekordbox (ADR-004).
"""
from __future__ import annotations

import logging
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path

from relocate_core import BaseRelocateWorker, RelocateScope, scope_contains
from rekordbox_fixes import FixStore
from rekordbox_xml import RekordboxXmlLibrary, get_playlist_tree, open_library

_log = logging.getLogger("listBuddy")
_PROGRESS_EVERY = 25


@dataclass
class XmlBrokenTrack:
    title: str
    artist: str
    original_path: Path
    location: str


def _iter_tracks(nodes: list[dict]):
    for node in nodes:
        pl = node["playlist"]
        if pl.is_folder:
            yield from _iter_tracks(node["children"])
        else:
            yield from pl.tracks


def find_broken_tracks(
    library: RekordboxXmlLibrary,
    should_stop: Callable[[], bool] | None = None,
    on_progress: Callable[[int, int], None] | None = None,
    scope: RelocateScope | None = None,
) -> list[tuple[str, XmlBrokenTrack]]:
    """Pistas de las playlists cuyo archivo (ya con reparaciones aplicadas) no
    existe. Una por `Location`, aunque esté en varias playlists."""
    unique: dict[str, object] = {}
    for t in _iter_tracks(get_playlist_tree(library)):
        if t.location and t.location not in unique:
            unique[t.location] = t
    tracks = list(unique.values())
    broken: list[tuple[str, XmlBrokenTrack]] = []
    total = len(tracks)
    for i, t in enumerate(tracks, start=1):
        if should_stop is not None and should_stop():
            break
        if scope_contains(scope, t.FolderPath) and not t.file_path.exists():
            broken.append((t.location, XmlBrokenTrack(t.title, t.artist, t.file_path, t.location)))
        if on_progress is not None and (i % _PROGRESS_EVERY == 0 or i == total):
            on_progress(i, total)
    return broken


class RekordboxXmlRelocateWorker(BaseRelocateWorker):
    def __init__(
        self, xml_path: Path, fixes_path: Path, search_root: Path,
        auto_resolve: bool = False, scope: RelocateScope | None = None,
    ) -> None:
        super().__init__(search_root, auto_resolve=auto_resolve, scope=scope)
        self._xml_path = xml_path
        self._store = FixStore(fixes_path)
        self._new_fixes: dict[str, str] = {}

    def _broken_scope_noun(self) -> str:
        return "lista exportada de Rekordbox"

    def _write_target_name(self) -> str:
        return "XML"

    def _engine_name(self) -> str:
        return "Rekordbox XML"

    def _label_for(self, broken_track: XmlBrokenTrack) -> str:
        return broken_track.title or broken_track.original_path.name

    def _apply_one(self, raw_entry: str, broken_track: XmlBrokenTrack, chosen: Path) -> bool:
        self._new_fixes[raw_entry] = str(chosen)
        return True

    def run(self) -> None:
        _log.info("Relocate Rekordbox XML: inicio · xml=%s · search_root=%s · auto=%s",
                  self._xml_path, self._search_root, self._auto_resolve)
        try:
            library = open_library(self._xml_path, fixes=self._store.load())
        except RuntimeError as e:
            self.log.emit(f"✗  {e}")
            self.finished_ok.emit(0, 0, 0, "error")
            return

        self.log.emit("🔎  Detectando enlaces rotos…")
        broken = find_broken_tracks(
            library,
            should_stop=self.isInterruptionRequested,
            on_progress=lambda done, total: self.progress.emit(done, total),
            scope=self._scope,
        )
        if self.isInterruptionRequested():
            self._emit_cancelled(0, 0, 0)
            return

        result = self._run_resolution(broken)
        if result is None:
            return
        if result.repaired == 0:
            self.log.emit("\nSin reparaciones para guardar.")
            self.finished_ok.emit(0, result.skipped, result.unresolved, "ok")
            return
        try:
            self._store.merge(self._new_fixes)
        except OSError as e:
            self.log.emit("✗  No se pudieron guardar las reparaciones (espacio o permisos).")
            _log.error("Relocate Rekordbox XML: no se pudo guardar %s: %s", self._store.path, e)
            self.finished_ok.emit(0, 0, 0, "error")
            return
        self.log.emit(
            "💾  Reparaciones guardadas en List Buddy. Tu XML y Rekordbox no se modificaron."
        )
        self._log_final_success(result.repaired, result.skipped, result.unresolved)
```

```python
@dataclass
class WriteResult:
    playlists: int
    tracks: int
    relocated: int
    missing: int


def write_playlists_xml(
    playlists: list[RbxPlaylist], dest: Path, *, source: Path | None = None,
    name_suffix: str = " (List Buddy)", app_version: str = "",
) -> WriteResult:
    """XML nuevo (formato oficial) con solo estas playlists, para importar en
    Rekordbox. Copia cada TRACK original (cues, beatgrid, metadatos) y pone en
    `Location` la ruta reparada. Nunca escribe sobre el XML de origen."""
    if source is not None and dest.resolve() == source.resolve():
        raise ValueError("No se puede sobrescribir el XML que exportaste desde Rekordbox.")
    root = ET.Element("DJ_PLAYLISTS", Version="1.0.0")
    ET.SubElement(root, "PRODUCT", Name="List Buddy", Version=app_version, Company="")
    collection = ET.SubElement(root, "COLLECTION")
    ids: dict[str, str] = {}
    relocated = missing = 0
    for pl in playlists:
        for t in pl.tracks:
            if t.location in ids:
                continue
            new_id = str(len(ids) + 1)
            ids[t.location] = new_id
            el = copy.deepcopy(t.element) if t.element is not None else ET.Element(
                "TRACK", Name=t.title, Artist=t.artist)
            el.set("TrackID", new_id)
            el.set("Location", path_to_location(t.file_path) if t.relocated else t.location)
            collection.append(el)
            relocated += int(t.relocated)
            missing += int(not t.file_path.exists())
    collection.set("Entries", str(len(ids)))
    root_node = ET.SubElement(ET.SubElement(root, "PLAYLISTS"), "NODE",
                              Type="0", Name="ROOT", Count=str(len(playlists)))
    for pl in playlists:
        node = ET.SubElement(root_node, "NODE", Name=pl.name + name_suffix, Type="1",
                             KeyType="0", Entries=str(len(pl.tracks)))
        for t in pl.tracks:
            ET.SubElement(node, "TRACK", Key=ids[t.location])
    ET.indent(root)
    dest.parent.mkdir(parents=True, exist_ok=True)
    tmp = dest.with_name(dest.name + ".tmp")
    ET.ElementTree(root).write(tmp, encoding="UTF-8", xml_declaration=True)
    os.replace(tmp, dest)
    return WriteResult(len(playlists), len(ids), relocated, missing)
```
