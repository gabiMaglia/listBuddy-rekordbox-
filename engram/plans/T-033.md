```python
"""Corre snippets Qt en un intérprete aparte.

Mientras PyQt6 siga instalado, importar un módulo del proyecto y PySide6 en el
mismo proceso cargaría dos copias de las DLL de Qt 6.11.1: el subproceso lo evita
y además mantiene limpio el proceso de pytest.
"""
from __future__ import annotations

import os
import subprocess
import sys
import textwrap
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent


def run_isolated(code: str, timeout: float = 60.0) -> subprocess.CompletedProcess[str]:
    env = {
        **os.environ,
        "QT_QPA_PLATFORM": "offscreen",
        "PYTHONPATH": str(PROJECT_ROOT),
    }
    return subprocess.run(
        [sys.executable, "-c", textwrap.dedent(code)],
        cwd=PROJECT_ROOT,
        env=env,
        capture_output=True,
        text=True,
        timeout=timeout,
    )
```

```python
"""T-033: el código propio usa PySide6 (LGPL), nunca PyQt6 (GPL) — ADR-003."""
from __future__ import annotations

import ast
from pathlib import Path

import pytest

from _qt_isolated import PROJECT_ROOT, run_isolated

_EXCLUDED_DIRS = {".venv", "venv", "build", "dist", ".git", "__pycache__"}


def _project_py_files() -> list[Path]:
    return [
        p for p in PROJECT_ROOT.rglob("*.py")
        if not _EXCLUDED_DIRS.intersection(p.relative_to(PROJECT_ROOT).parts)
    ]


def _imported_roots(path: Path) -> set[str]:
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    roots: set[str] = set()
    # ast.walk incluye imports diferidos dentro de funciones (app_logging, ui).
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            roots.update(a.name.split(".")[0] for a in node.names)
        elif isinstance(node, ast.ImportFrom) and node.level == 0 and node.module:
            roots.add(node.module.split(".")[0])
    return roots


def test_no_project_file_imports_pyqt6() -> None:
    offenders = sorted(
        str(p.relative_to(PROJECT_ROOT))
        for p in _project_py_files()
        if "PyQt6" in _imported_roots(p)
    )
    assert offenders == []


# (módulo, clase, señal) — una por cada archivo que declara señales.

    ("worker", "ExportWorker", "log"),
    ("preview_worker", "PreviewWorker", "group_ready"),
    ("relocate_core", "RestoreBackupWorker", "finished_ok"),

        import {module} as m

        print("IS_PYSIDE_SIGNAL" if isinstance(sig, Signal) else type(sig).__module__)

    assert proc.returncode == 0, proc.stderr
```

```python
from __future__ import annotations

from _qt_isolated import run_isolated

_PRELUDE = """
    import numpy as np
    from PySide6.QtCore import QByteArray
    from PySide6.QtMultimedia import QAudioBuffer, QAudioFormat
    import spectro_worker as sw

    def make(raw, fmt, ch):
        f = QAudioFormat()
        f.setSampleFormat(fmt)
        f.setChannelCount(ch)
        f.setSampleRate(11025)
        return QAudioBuffer(QByteArray(raw), f)

    SF = QAudioFormat.SampleFormat
"""


def _run(body: str) -> str:
    proc = run_isolated(_PRELUDE + body)
    assert proc.returncode == 0, proc.stderr
    return proc.stdout.strip()


def test_int16_mono_is_normalized() -> None:
    out = _run("""
    raw = np.array([16384, -16384], np.int16).tobytes()
    print(sw._buffer_to_mono(make(raw, SF.Int16, 1)).tolist())
    """)
    assert out == "[0.5, -0.5]"


def test_int16_stereo_is_downmixed() -> None:
    out = _run("""
    raw = np.array([16384, 0, -16384, 0], np.int16).tobytes()
    print(sw._buffer_to_mono(make(raw, SF.Int16, 2)).tolist())
    """)
    assert out == "[0.25, -0.25]"


def test_uint8_offset_is_removed() -> None:
    out = _run("""
    raw = np.array([192, 64], np.uint8).tobytes()
    print(sw._buffer_to_mono(make(raw, SF.UInt8, 1)).tolist())
    """)
    assert out == "[0.25, -0.25]"
```
