def _themes() -> dict[str, dict[str, str]]:
    r = run_isolated("import json, styles; print(json.dumps(styles.THEMES))")
    assert r.returncode == 0, r.stderr
    return json.loads(r.stdout)


def _lum(hex_color: str) -> float:
    h = hex_color.lstrip("#")
    chans = [int(h[i:i + 2], 16) / 255 for i in (0, 2, 4)]
    lin = [c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4 for c in chans]
    return 0.2126 * lin[0] + 0.7152 * lin[1] + 0.0722 * lin[2]


def ratio(fg: str, bg: str) -> float:
    hi, lo = sorted((_lum(fg), _lum(bg)), reverse=True)
    return (hi + 0.05) / (lo + 0.05)


def test_ratio_function_is_not_vacuous() -> None:
    assert round(ratio("#000000", "#ffffff"), 1) == 21.0
    assert ratio("#777777", "#808080") < MIN_RATIO  # un par malo SÍ se detecta


def test_new_tokens_match_spec_exactly() -> None:
    themes = _themes()
    for name, expected in EXPECTED.items():
        got = {k: themes[name][k] for k in expected}
        assert got == expected


def test_all_text_pairs_meet_wcag() -> None:
    themes = _themes()
    fallas: list[str] = []
    for name in EXPECTED:
        t = themes[name]
        pairs = [(f, g) for f in TEXT_TOKENS for g in GROUNDS] + list(EXTRA_PAIRS)
        for fg, bg in pairs:
            r = ratio(t[fg], t[bg])
            if r < MIN_RATIO:
                fallas.append(f"{name}: {fg} sobre {bg} = {r:.2f}")
    assert fallas == []  # enumera la diferencia, no ejemplos a mano

FONTS_DIR = ROOT / 'fonts'
_log = logging.getLogger(__name__)


def register_fonts() -> list[str]:
    """Registra las fuentes OFL bundleadas; sin ellas la app arranca con fallback."""
    families: set[str] = set()
    for path in sorted(FONTS_DIR.glob('*.ttf')):
        font_id = QFontDatabase.addApplicationFont(str(path))
        if font_id == -1:
            _log.warning('No se pudo cargar la fuente %s', path.name)
            continue
        families.update(QFontDatabase.applicationFontFamilies(font_id))
    if not families:
        _log.warning('Sin fuentes bundleadas en %s; se usa el fallback del sistema', FONTS_DIR)
    return sorted(families)

    if n <= 0:
        return ""
    return "1 roto" if n == 1 else f"{n} rotos"


class MaintenanceButton(QPushButton):
    """Botón del header: ícono + «Mantenimiento» + badge de rotos.

    Los hijos son QLabel transparentes al mouse para que todo el botón sea
    clickeable; sizeHint sale del layout porque QPushButton ignora sus hijos.
    """

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self.setObjectName("maint_btn")
        self.setCursor(Qt.CursorShape.PointingHandCursor)
        self.setFixedHeight(34)
        self.setAccessibleName("Mantenimiento")

        lo.setContentsMargins(12, 0, 12, 0)
        lo.setSpacing(8)

        self._text = QLabel("Mantenimiento")
        self._text.setObjectName("maint_text")
        self._badge = QLabel()
        self._badge.setObjectName("maint_badge")
        self._badge.setVisible(False)
        for w in (self._icon, self._text, self._badge):
            w.setAttribute(Qt.WidgetAttribute.WA_TransparentForMouseEvents, True)
            lo.addWidget(w)

    def icon_label(self) -> QLabel:
        return self._icon

    def set_broken(self, n: int) -> None:
        txt = broken_badge_text(n)
        self._badge.setText(txt)
        self._badge.setVisible(bool(txt))
        self.setAccessibleDescription(txt)

    def set_compact(self, compact: bool) -> None:
        self._text.setVisible(not compact)
        self.setToolTip("Mantenimiento" if compact else "")

    def sizeHint(self) -> QSize:
        return self.layout().sizeHint()

    def minimumSizeHint(self) -> QSize:
        return self.layout().sizeHint()
