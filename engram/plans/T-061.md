def _themes() -> dict[str, dict[str, str]]:
    r = run_isolated("import json, styles; print(json.dumps(styles.THEMES))")
    assert r.returncode == 0, r.stderr
    return json.loads(r.stdout)


def _lum(hex_color: str) -> float:
    h = hex_color.lstrip("#")
    chans = [int(h[i:i + 2], 16) / 255 for i in (0, 2, 4)]
    lin = [c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4 for c in chans]
    return 0.2126 * lin[0] + 0.7152 * lin[1] + 0.0722 * lin[2]


def ratio(fg: str, bg: str) -> float:
    hi, lo = sorted((_lum(fg), _lum(bg)), reverse=True)
    return (hi + 0.05) / (lo + 0.05)


def test_ratio_function_is_not_vacuous() -> None:
    assert round(ratio("#000000", "#ffffff"), 1) == 21.0
    assert ratio("#777777", "#808080") < MIN_RATIO  # un par malo SÍ se detecta


def test_new_tokens_match_spec_exactly() -> None:
    themes = _themes()
    for name, expected in EXPECTED.items():
        got = {k: themes[name][k] for k in expected}
        assert got == expected


def test_all_text_pairs_meet_wcag() -> None:
    themes = _themes()
    fallas: list[str] = []
    for name in EXPECTED:
        t = themes[name]
        pairs = [(f, g) for f in TEXT_TOKENS for g in GROUNDS] + list(EXTRA_PAIRS)
        for fg, bg in pairs:
            r = ratio(t[fg], t[bg])
            if r < MIN_RATIO:
                fallas.append(f"{name}: {fg} sobre {bg} = {r:.2f}")
    assert fallas == []  # enumera la diferencia, no ejemplos a mano

FONTS_DIR = ROOT / 'fonts'
_log = logging.getLogger(__name__)


def register_fonts() -> list[str]:
    """Registra las fuentes OFL bundleadas; sin ellas la app arranca con fallback."""
    families: set[str] = set()
    for path in sorted(FONTS_DIR.glob('*.ttf')):
        font_id = QFontDatabase.addApplicationFont(str(path))
        if font_id == -1:
            _log.warning('No se pudo cargar la fuente %s', path.name)
            continue
        families.update(QFontDatabase.applicationFontFamilies(font_id))
    if not families:
        _log.warning('Sin fuentes bundleadas en %s; se usa el fallback del sistema', FONTS_DIR)
    return sorted(families)

```python
"""
waveform_worker.py
------------------
Picos de volumen de un tema completo para la onda del footer.

Decodifica con QAudioDecoder (mismo backend que la reproducción, mismos formatos),
guarda SOLO el pico absoluto de cada ventana de ~50 ms (no el audio) y emite la lista
normalizada. Corre en un QThread con su propio event loop: la UI nunca se bloquea.
A diferencia del espectrograma que reemplaza no hay tope de 90 s: la onda tiene que
representar el tema entero para que el click haga seek al lugar correcto.
"""
from __future__ import annotations

import numpy as np
from PySide6.QtCore import QEventLoop, QThread, QTimer, QUrl, Signal
from PySide6.QtMultimedia import QAudioDecoder, QAudioFormat

from audio_decode import buffer_to_mono

TARGET_SR = 11025
WIN = 551                      # ≈ 50 ms a 11025 Hz
MAX_SECONDS = 20 * 60          # tope de seguridad (sets larguísimos)
DECODE_TIMEOUT = 90_000        # ms; corta si el decode se cuelga


def window_peaks(
    samples: np.ndarray, carry: np.ndarray, win: int = WIN
) -> tuple[np.ndarray, np.ndarray]:
    """Pico absoluto por ventana; lo que no completa una ventana vuelve como `carry`."""
    mag = np.abs(samples)
    data = np.concatenate([carry, mag]) if carry.size else mag
    n = data.size // win
    if n == 0:
        return np.empty(0, np.float32), data
    peaks = data[: n * win].reshape(n, win).max(axis=1).astype(np.float32)
    return peaks, data[n * win:]


def normalize_peaks(peaks: np.ndarray) -> list[float]:
    """0..1 con techo en el percentil 99: un click aislado no aplasta el resto."""
    if peaks.size == 0:
        return []
    ceiling = float(np.percentile(peaks, 99))
    if ceiling <= 1e-6:
        return [0.0] * int(peaks.size)
    return np.clip(peaks / ceiling, 0.0, 1.0).tolist()


class WaveformWorker(QThread):
    ready: Signal = Signal(object, int)   # (picos normalizados, generation)

    def __init__(self, path: str, generation: int) -> None:
        super().__init__()
        self.path = path
        self._gen = generation
        self._stop = False
        self._parts: list[np.ndarray] = []
        self._carry = np.empty(0, np.float32)
        self._samples = 0

    def cancel(self) -> None:
        self._stop = True

    def run(self) -> None:
        try:
            self._decode()
            if self._stop or not self._parts:
                return
            peaks = normalize_peaks(np.concatenate(self._parts))
            if peaks and not self._stop:
                self.ready.emit(peaks, self._gen)
        except Exception:
            return  # sin onda, sin ruido: el player sigue funcionando

    def _decode(self) -> None:
        decoder = QAudioDecoder()
        fmt = QAudioFormat()
        fmt.setSampleRate(TARGET_SR)
        fmt.setChannelCount(1)
        fmt.setSampleFormat(QAudioFormat.SampleFormat.Int16)
        decoder.setAudioFormat(fmt)
        decoder.setSource(QUrl.fromLocalFile(self.path))
        loop = QEventLoop()
        limit = TARGET_SR * MAX_SECONDS

        def on_buffer() -> None:
            while decoder.bufferAvailable():
                mono = buffer_to_mono(decoder.read())
                if mono is not None and mono.size:
                    peaks, self._carry = window_peaks(mono, self._carry)
                    if peaks.size:
                        self._parts.append(peaks)
                    self._samples += mono.size
                if self._stop or self._samples >= limit:
                    decoder.stop()
                    loop.quit()
                    return

        decoder.bufferReady.connect(on_buffer)
        decoder.finished.connect(loop.quit)
        QTimer.singleShot(DECODE_TIMEOUT, loop.quit)
        decoder.start()
        loop.exec()
        decoder.stop()
```

```python
def bar_heights(peaks: Sequence[float], n: int, floor: float = 0.12) -> list[float]:
    """n alturas 0..1 por max-pool de los picos; sin picos, barras planas al piso."""
    if n <= 0:
        return []
    if not peaks:
        return [floor] * n
    total = len(peaks)
    out: list[float] = []
    for i in range(n):
        a = i * total // n
        b = max(a + 1, (i + 1) * total // n)
        out.append(max(floor, max(peaks[a:b])))
    return out


class WaveformSeek(SeekBar):
    """SeekBar con forma de onda: mismo click/arrastre, barras en vez de riel."""

    _BAR, _GAP = 3, 2

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self.setFixedHeight(32)
        self._peaks: Sequence[float] = ()
        self._rest = QColor(styles.token("dark", "line"))

    def set_peaks(self, peaks: Sequence[float] | None) -> None:
        self._peaks = peaks or ()
        self.update()

    def set_colors(self, played_hex: str, rest_hex: str) -> None:
        self._accent = QColor(played_hex)
        self._rest = QColor(rest_hex)
        self.update()

    def paintEvent(self, _event) -> None:
        p = QPainter(self)
        p.setRenderHint(QPainter.RenderHint.Antialiasing, False)
        step = self._BAR + self._GAP
        n = max(1, (self.width() + self._GAP) // step)
        heights = bar_heights(self._peaks, n)
        frac = (self._pos / self._dur) if self._dur > 0 else 0.0
        mid = self.height() / 2
        p.setPen(Qt.PenStyle.NoPen)
        for i, level in enumerate(heights):
            x = i * step
            h = max(2.0, level * (self.height() - 2))
            # La barra cuenta como reproducida si su centro ya fue pasado.
            p.setBrush(self._accent if (x + self._BAR / 2) / self.width() <= frac else self._rest)
            p.drawRect(QRectF(x, mid - h / 2, self._BAR, h))
```
